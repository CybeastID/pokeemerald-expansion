MysteryZone_MapScripts::
	map_script MAP_SCRIPT_ON_TRANSITION, OnTransition
	map_script MAP_SCRIPT_ON_FRAME_TABLE, On_Frame
	.byte 0

On_Frame:
	map_script_2 VAR_CASTER_REVEAL, 1, OnFrame_1
	.2byte 0

OnFrame_1::
	delay 10
	msgbox CutieTrap_Text
	waitmessage
	closemessage
	callnative StartPlayerShrinkAnimation
	fadenewbgm MUS_ENCOUNTER_SUSPICIOUS
	delay 70
	msgbox OnTransition_Text_0
	waitmessage
	closemessage
	addvar VAR_CASTER_REVEAL, 1
	fadescreenspeed FADE_TO_BLACK, 1
	fadeoutbgm 1
	msgbox CutieTrap2_Text
	waitmessage
	closemessage
	releaseall
	warpwhitefade MAP_CASTERS_TERRARIUM, 6, 5
	playse SE_FALL
	return
	
	
	waitmessage
	closemessage

Dbug1_Text:
	.string "Debug: Player should be falling now.$"

CutieTrap_Text:
	.string "What a delightfully cute pet I've\n"
	.string "found! They'd look positively fetching\l"
	.string "in my terrarium!$"
	
OnTransition::
	compare VAR_CASTER_REVEAL, 0
	goto_if_eq OnTransition_1
	compare VAR_CASTER_REVEAL, 1
	goto_if_eq OnTransition_4
	return

OnTransition_1:
	setflag FLAG_HIDES_CASTER
	return

OnTransition_4:
	
	return


CastersTrap::
	msgbox CastersTrap_Text_0
	waitmessage
	closemessage
	setvar VAR_CASTER_REVEAL, 1
	clearflag FLAG_HIDES_CASTER
	warpteleport MAP_MYSTERY_ZONE, 2
	fadescreen FADE_FROM_BLACK
	return


OnTransition_Text_0:
	.string "Teehee~! A little cutie like you...\n"
	.string "you'll never escape me! Well...in you go,\l"
	.string "my cute little pet Trainer!$"

CutieTrap2_Text:
	.string "The mysterious girl's hand closed around\n"
	.string "you! What's happening...?!$"


CastersTrap_Text_0:
	.string "{COLOR LIGHT_GRAY} ??? {COLOR DARK_GRAY}used Teleport!$"
